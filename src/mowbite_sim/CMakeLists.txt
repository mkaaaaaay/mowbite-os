  # Every node of a launch test writes its journal into a folder of the test (MOWBITE_JOURNAL_DIR); after the test the
  # journal has to replay bit-identically. Folder cleared before, replay after: ctest fixtures keep the order.
  function(add_journaled_launch_test file)
    cmake_parse_arguments(ARG "" "TARGET;TIMEOUT" "ENV" ${ARGN})
    if(NOT ARG_TARGET)
      get_filename_component(ARG_TARGET ${file} NAME_WE)
    endif()
    set(dir "${CMAKE_CURRENT_BINARY_DIR}/journal/${ARG_TARGET}")
    add_test(NAME ${ARG_TARGET}_journal_clear COMMAND ${CMAKE_COMMAND} -E rm -rf ${dir})
    add_launch_test(${file} TARGET ${ARG_TARGET} TIMEOUT ${ARG_TIMEOUT}
      ENV ${ARG_ENV} MOWBITE_JOURNAL_DIR=${dir})
    add_test(NAME ${ARG_TARGET}_replay COMMAND ${MOWBITE_REPLAY} check --if-any ${dir})
    set_tests_properties(${ARG_TARGET}_journal_clear PROPERTIES FIXTURES_SETUP ${ARG_TARGET}_clear)
    set_tests_properties(${ARG_TARGET} PROPERTIES FIXTURES_REQUIRED ${ARG_TARGET}_clear
                                                 FIXTURES_SETUP ${ARG_TARGET}_journal)
    set_tests_properties(${ARG_TARGET}_replay PROPERTIES FIXTURES_REQUIRED ${ARG_TARGET}_journal)
  endfunction()
